CREATE OR REPLACE FUNCTION settype()
  RETURNS TRIGGER 
AS
$$
BEGIN
	UPDATE device SET typeid = 8
	WHERE typeid = OLD.typeid;
	RETURN NULL;
END; $$ LANGUAGE PLPGSQL;


CREATE OR REPLACE TRIGGER deleteType
  BEFORE DELETE
  ON devicetype
  FOR EACH ROW
  EXECUTE PROCEDURE settype();

CREATE OR REPLACE FUNCTION setDate()
  RETURNS TRIGGER 
AS
$$
BEGIN
	IF NEW.receiptdata <> now()::DATE THEN
		UPDATE offer SET receiptdata = now()::DATE WHERE offerid=NEW.offerid;
	END IF;

	RETURN NEW;
END; $$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER trueDate
  AFTER INSERT
  ON offer
  FOR EACH ROW
  EXECUTE PROCEDURE setDate();

CREATE OR REPLACE FUNCTION setQualification()
  RETURNS TRIGGER 
AS
$$
BEGIN
	IF NEW.qualification < OLD.qualification THEN 
		NEW.qualification:=OLD.qualification;        
	END IF;
	RETURN NEW;
END; $$ LANGUAGE PLPGSQL;

CREATE OR REPLACE TRIGGER updateQualification
  BEFORE UPDATE
  ON master
  FOR EACH ROW
  EXECUTE PROCEDURE setQualification();
