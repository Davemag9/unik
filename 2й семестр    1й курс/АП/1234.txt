#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define EPS 1e-7
ld coef=M_PI*4.0/3.0;

ld VSeg(ld h,ld r){ return M_PI * pow(h,2) * (r - h / 3.0); }

ld VSfer(ld r){ return coef*pow(r,3); }

ld VCObol(ld d, ld r, ld h){ return VSeg(h, r) - VSeg(h-d, r-d); }

ld cost1(ld h, ld d, ld r, ld c){
    //cout<<1<<endl;
    return (VSeg(h, r) * c) / 1.0;
}
ld cost2(ld h, ld d, ld r, ld c){
    //cout<<2<<endl;
    return (VSeg(h-d,r-d) + VCObol(d, r, h) * c) / 1.0;
}
ld cost3(ld h, ld d, ld r, ld c){
    //cout<<3<<endl;
    return (VSfer(r-d) + (VSeg(h, r) - VSfer(r-d)) * c) / 1.0;
}

ld cost(ld h, ld r, ld d, ld c){
    if(h <= d) return cost1(h, d, r, c);
    if(h >= (2*r - d)) return cost3(h, d, r, c);
    return cost2(h, d, r, c);
}

int main() {
    cout.precision(7);
    ld r, d, c, k;
    //r=3.0;d=1.0;c=2.0;k=2.0;
    cin >> r >> d >> c >> k;
    ld left =  r * (-1.0), right = r * 1.0, mid;
    while ((right - left) >= EPS){
        mid = (ld) ((left +  right)) / 2.0;
        //cout<<left<<" "<<mid<<" "<<right<<endl;
        //cout << cost(r-mid, r, d, c) << endl;
        if(cost(r-mid, r, d, c) >= k) left = mid ;
        else right = mid;
    }
    cout<<fixed<<left;
}

#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double

ld lw, rw, r, n;
vector<ld> x, y; //x - first y - last
vector<vector<ll>> graph;
vector<bool> visited;

void builDer(ll i, ll j){
    graph[i].push_back(j);
    graph[j].push_back(i);
}

void dfs(ll v){

    visited[v] = true;
    for(auto x: graph[v]){
        if(!visited[x]){
            dfs(x);
        }
    }
}

bool check(ld mid){
    for(ll i = 0; i < n + 2; i++) graph[i].clear();
    for(ll i = 0 ; i < n; i++){
        if (mid >= abs(x[i] - lw) - r) builDer(i, n);
        if (mid >= abs(x[i] - rw) - r) builDer(i, n + 1);
        for(ll j = i + 1; j < n; j++){
            if (mid >= (sqrt(pow(x[i] - x[j], 2) + pow(y[i] - y[j], 2)) - 2 * r)) builDer(i, j);
        }
    }
    visited.assign(n + 2, false);
    dfs(n);
    return visited[n + 1];
}



int main(){
    cin >> lw >> rw >> r >> n;
    x.resize(n, 0);
    y.resize(n, 0);
    visited.resize(n + 2);
    graph.resize(n + 2);
    for (ll i = 0; i < n; i++) cin >> x[i] >> y[i];
    ld left = 0, right = 1e9 + 1;
    while ((right - left) > 1e-5){
        ld mid = (right + left) / 2;
        if(!check(mid)) left = mid;
        else right = mid;
    }
    cout.precision(4);
    cout << fixed << right << endl;
}

int main(){
    ll n;
    cin >> n;

    visited.assign(n, false);
    graph.resize(n);
    vector<pair<ld, ld>> coords;
    vector<ld> rad;

    for(ll i = 0; i < n; i++){
        ld a, b, c;
        cin >> a >> b >> c;
        coords.push_back({a, b});
        rad.push_back(c);
    }

    for(ll i  = 0; i < n; i++){
        for(ll j = i + 1; j < n; j++){
            if(sqrt(pow(coords[i].first - coords[j].first, 2) + pow(coords[i].second - coords[j].second, 2)) <= rad[i] + rad [j]){
                builDer(i, j);
            }
        }
    }

    for(ll i = 0; i < n; i++) {
        if(!visited[i]) {
            dfs(i);
            counter++;
        }
