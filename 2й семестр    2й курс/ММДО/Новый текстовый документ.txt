#include "bits/stdc++.h"

using namespace std;
vector<long double> C;
vector<long double> Plan;
vector<long double> Cb;
vector<string> Basis;
vector<string> zminni;
vector<vector<long double>> matrix;
vector<vector<long double>> Z;
vector<long double> Xb_to_Xi;

map<string, int> zminna_to_int;
long double M = INT_MAX;

int main() {
    C = {-2, 1, 0, 0, 0, M, M};
    Plan = {8, 6, 3};
    Cb = {0, M, M};
    zminni = {"x1", "x2", "x3", "x4", "x5", "x6", "x7"};
    Basis = {"x3", "x6", "x7"};

    matrix.resize(Basis.size());
    matrix[0] = {2, 1, 1, 0, 0, 0, 0};
    matrix[1] = {1, 3, 0, -1, 0, 1, 0};
    matrix[2] = {3, 1, 0, 0, -1, 0, 1};

    cout << "C: ";
    for (auto i: C) cout << i << " ";
    cout << endl;

    cout << "Plan: ";
    for (auto i: Plan) cout << i << " ";
    cout << endl;

    cout << "Cb: ";
    for (auto i: Cb) cout << i << " ";
    cout << endl;

    for (int i = 0; i < zminni.size(); i++) zminna_to_int[zminni[i]] = i;

    cout << "Basis: " << endl;
    for (auto i: Basis) cout << i << " : " << zminna_to_int[i] << endl;

    for (auto i: zminni) cout << zminna_to_int[i] << " ";
    cout << endl << "-----------------" << endl;

    for (int j = 0; j < Basis.size(); j++) {
        for (int i = 0; i < zminni.size(); i++) {
            cout << matrix[j][i] << " ";
        }
        cout << endl;
    }

    cout << endl;

    while (true){
        Z.assign(2, vector<long double>(zminni.size()));

        for (int i = 0; i < zminni.size(); i++) {
            for (int j = 0; j < Basis.size(); j++) {
                if (Cb[j] == M) {
                    Z[1][i] += matrix[j][i];
                } else {
                    Z[0][i] += Cb[j] * matrix[j][i];
                }
            }
            if (C[i] == M) {
                Z[1][i] -= 1;
            } else {
                Z[0][i] -= C[i];
            }
        }

        int c = 0;
        for(int i = 0; i < zminni.size(); i++){
            if(Z[1][i] * M + Z[0][i] <= 0) c++;
        }

        if(c == zminni.size()){
            cout << endl << "Optimal solution" << endl;
            long double f = 0;
            for(int i = 0; i < Basis.size(); i++){
                f += Cb[i] * Plan[i];
            }
            cout << "F = " << f << endl;
            return 0;
        }

        cout << endl;
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < zminni.size(); j++) {
                cout << Z[i][j] << " ";
            }
            cout << endl;
        }

        int maxcol = 0;
        for (int i = 0; i < zminni.size(); i++) {
            if (Z[1][maxcol] == Z[1][i]) {
                if (Z[0][maxcol] <= Z[0][i]) maxcol = i;
            } else if (Z[1][maxcol] < Z[1][i]) {
                maxcol = i;
            }
        }

        c = 0;
        for(int i = 0; i < Basis.size(); i++) {
            if(matrix[i][maxcol] <= 0) c++;
        }

        if(c == Basis.size()){
            cout << "No solution" << endl;
            return 0;
        }

        cout << "MAXCOL: " << maxcol << endl;

        Xb_to_Xi.resize(Basis.size());

        for (int i = 0; i < Basis.size(); i++) {
            if (matrix[i][maxcol] > 0) Xb_to_Xi[i] = Plan[i] / matrix[i][maxcol];
            else Xb_to_Xi[i] = INT_MAX;
        }

        int minrow;
        for (int i = 0; i < Basis.size(); i++) {
            if (Xb_to_Xi[i] > 0) minrow = i;
            break;
        }

        for (int i = 0; i < Basis.size(); i++) {
            if (Xb_to_Xi[i] < Xb_to_Xi[minrow] and Xb_to_Xi[i] >= 0) minrow = i;
        }

        cout << "MINROW: " << minrow << endl;
        cout << "RZV: " << matrix[minrow][maxcol] << endl;

        vector<long double> Plan_new;
        vector<long double> Cb_new;
        vector<string> Basis_new;
        vector<vector<long double>> matrix_new;

        Basis_new = Basis;

        for (int i = 0; i < Basis.size(); i++) {
            if (Basis_new[i] == Basis[minrow]) {
                Basis_new[i] = zminni[maxcol];
            }
        }

        cout << "Basis_new: " << endl;
        for (auto i: Basis_new) cout << i << " : " << zminna_to_int[i] << endl;

        Plan_new.resize(Basis.size());
        Cb_new.resize(Basis.size());
        matrix_new.resize(Basis.size());

        for (int i = 0; i < Basis.size(); i++) {
            matrix_new[i].assign(zminni.size(), 0);
        }

        for (int i = 0; i < Basis.size(); i++) {
            matrix_new[i][zminna_to_int[Basis_new[i]]] = 1;
        }

        for (int i = 0; i < Basis.size(); i++) {
            Cb_new[i] = C[zminna_to_int[Basis_new[i]]];
        }

        cout << "Cb_new: ";
        for (auto i: Cb_new) i == M ? cout << "M " : cout << i << " ";
        cout << endl;


        for (int i = 0; i < Basis.size(); i++) {
            if (i == minrow) Plan_new[i] = Plan[i] / matrix[i][maxcol];
            else Plan_new[i] = Plan[i] - Plan[minrow] * matrix[i][maxcol] / matrix[minrow][maxcol];
        }

        cout << "Plan_new: ";
        for (auto i: Plan_new) cout << i << " ";

        map<string, bool> tmp;
        for (auto i: Basis_new) {
            tmp[i] = true;
        }

        for (int i = 0; i < zminni.size(); i++) {
            if (!tmp[zminni[i]]) {
                for (int j = 0; j < Basis.size(); j++) {
                    if (j == minrow) matrix_new[j][i] = matrix[j][i] / matrix[minrow][maxcol];
                    else
                        matrix_new[j][i] =
                                matrix[j][i] - matrix[minrow][i] * matrix[j][maxcol] / matrix[minrow][maxcol];
                }
            }
        }

        Basis = Basis_new;
        Plan = Plan_new;
        Cb = Cb_new;
        matrix = matrix_new;
    }


}
